```go Go
import (
	"context"
	"fmt"

	lago "github.com/getlago/lago-go-client"
)

func main() {
	lagoClient := lago.New().SetApiKey("__YOUR_API_KEY__")
	ctx := context.Background()

	order, err := lagoClient.Order().Get(ctx, "__ORDER_ID__")
	if err != nil {
		// Error is *lago.Error
		panic(err)
	}

	// order is *lago.Order
	fmt.Println(order)
}
```
