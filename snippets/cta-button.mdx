export const CtaButton = ({ href, label, variant = "primary" }) => {
  const base = {
    display: "inline-flex",
    alignItems: "center",
    gap: "0.375rem",
    padding: "0.625rem 1.125rem",
    borderRadius: "0.5rem",
    fontFamily: "GT America",
    fontSize: "0.9375rem",
    fontWeight: 500,
    lineHeight: 1.2,
    textDecoration: "none",
    whiteSpace: "nowrap",
    transition: "opacity 0.15s ease, background-color 0.15s ease",
  };

  const styles =
    variant === "secondary"
      ? {
          ...base,
          border: "1px solid rgba(0, 108, 250, 0.35)",
          color: "#006CFA",
          backgroundColor: "transparent",
        }
      : {
          ...base,
          border: "1px solid #006CFA",
          color: "#FFFFFF",
          backgroundColor: "#006CFA",
        };

  return (
    <a
      href={href}
      className="not-prose lago-cta-button"
      style={styles}
      onMouseEnter={(e) => {
        e.currentTarget.style.opacity = "0.85";
      }}
      onMouseLeave={(e) => {
        e.currentTarget.style.opacity = "1";
      }}
    >
      {label}
    </a>
  );
};

export const CtaRow = ({ children }) => (
  <div
    className="not-prose"
    style={{
      display: "flex",
      flexWrap: "wrap",
      gap: "0.75rem",
      marginTop: "1.5rem",
      marginBottom: "0.5rem",
    }}
  >
    {children}
  </div>
);
